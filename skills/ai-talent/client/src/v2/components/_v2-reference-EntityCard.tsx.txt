import { Card } from "@/components/ui/card";
import { Button } from "@/components/ui/button";
import { Badge } from "@/components/ui/badge";
import { Trash2, Sparkles } from "lucide-react";

type EntityType = "brand" | "product" | "series";

type EntityStatus = "pending" | "analyzing" | "review" | "completed";

interface EntityCardProps {
  type: EntityType;
  entity: {
    id: number;
    name: string;
    status: EntityStatus;
    tagline?: string | null;
    philosophy?: string | null; // 品牌使用
    valueProposition?: string | null; // 產品/系列使用
  };
  onDelete: () => void;
  onViewPositioning: () => void;
  onViewReport: () => void;
  onAIAdvisor: () => void;
}

// 配色映射
const colorMap = {
  brand: {
    bg: "bg-slate-600",
    text: "text-slate-600",
    border: "border-slate-600",
    hover: "hover:bg-slate-700",
    light: "bg-slate-50",
  },
  product: {
    bg: "bg-lime-600",
    text: "text-lime-600",
    border: "border-lime-600",
    hover: "hover:bg-lime-700",
    light: "bg-lime-50",
  },
  series: {
    bg: "bg-violet-600",
    text: "text-violet-600",
    border: "border-violet-600",
    hover: "hover:bg-violet-700",
    light: "bg-violet-50",
  },
};

// 狀態映射
const statusMap = {
  pending: { label: "待處理", variant: "secondary" as const },
  analyzing: { label: "分析中", variant: "secondary" as const },
  review: { label: "審視中", variant: "default" as const },
  completed: { label: "已發佈", variant: "secondary" as const },
};

// 實體類型標籤映射
const entityLabels = {
  brand: {
    tagline: "品牌標語",
    taglineEn: "BRAND TAGLINE",
    content: "品牌哲學",
    contentEn: "BRAND PHILOSOPHY",
    positioning: "定位",
    report: "視覺規範",
    advisor: "AI 顧問",
  },
  product: {
    tagline: "產品標語",
    taglineEn: "PRODUCT TAGLINE",
    content: "價值主張",
    contentEn: "VALUE PROPOSITION",
    positioning: "定位",
    report: "行銷規範",
    advisor: "AI 顧問",
  },
  series: {
    tagline: "系列標語",
    taglineEn: "SERIES TAGLINE",
    content: "價值主張",
    contentEn: "VALUE PROPOSITION",
    positioning: "定位",
    report: "行銷規範",
    advisor: "AI 顧問",
  },
};

export function EntityCard({
  type,
  entity,
  onDelete,
  onViewPositioning,
  onViewReport,
  onAIAdvisor,
}: EntityCardProps) {
  const colors = colorMap[type];
  const labels = entityLabels[type];
  const statusInfo = statusMap[entity.status];

  // 獲取實體首字母
  const getInitial = (name: string) => {
    return name.charAt(0).toUpperCase();
  };

  // 判斷定位是否完成
  const isPositioningCompleted = entity.status === "completed";

  // 判斷視覺規範/行銷規範按鈕是否可用
  const isReportEnabled = type === "brand" ? isPositioningCompleted : true;

  return (
    <Card className="relative overflow-hidden border-2 hover:shadow-lg transition-shadow">
      {/* 卡片頂部 */}
      <div className="p-6 pb-4">
        <div className="flex items-start justify-between mb-4">
          <div className="flex items-center gap-3">
            {/* 實體圖標 */}
            <div
              className={`w-12 h-12 rounded-full ${colors.bg} flex items-center justify-center text-white font-bold text-xl`}
            >
              {getInitial(entity.name)}
            </div>
            
            {/* 實體名稱和狀態 */}
            <div>
              <h3 className="font-semibold text-lg">{entity.name}</h3>
              <Badge
                variant={statusInfo.variant}
                className={
                  entity.status === "review" ? `${colors.bg} text-white` : ""
                }
              >
                {statusInfo.label}
              </Badge>
            </div>
          </div>

          {/* 刪除按鈕 */}
          <Button
            variant="ghost"
            size="icon"
            onClick={onDelete}
            className="text-gray-400 hover:text-red-500"
          >
            <Trash2 className="h-4 w-4" />
          </Button>
        </div>

        {/* 內容區塊 */}
        <div className="space-y-4">
          {/* 標語區塊 */}
          <div className={`p-4 rounded-lg border ${colors.light}`}>
            <div className="text-xs font-semibold text-gray-500 mb-1">
              {labels.taglineEn} / {labels.tagline}
            </div>
            <div className="text-sm text-gray-700">
              {entity.tagline || "尚未設定"}
            </div>
          </div>

          {/* 哲學/價值主張區塊 */}
          <div className={`p-4 rounded-lg border ${colors.light}`}>
            <div className="text-xs font-semibold text-gray-500 mb-1">
              {labels.contentEn} / {labels.content}
            </div>
            <div className="text-sm text-gray-700">
              {type === "brand"
                ? entity.philosophy || "尚未設定"
                : entity.valueProposition || "尚未設定"}
            </div>
          </div>
        </div>
      </div>

      {/* 操作按鈕區 */}
      <div className="px-6 pb-6 flex gap-2">
        {/* 定位按鈕 */}
        <button
          onClick={onViewPositioning}
          className="flex-1 text-white font-semibold text-sm px-3 py-2 transition-all hover:opacity-90 hover:-translate-y-0.5"
          style={{
            background: "linear-gradient(135deg, #EA580C, #F97316)",
            borderRadius: "50px",
            border: "none",
            boxShadow: "0 2px 8px rgba(234,88,12,0.25)",
          }}
        >
          {isPositioningCompleted ? "查看定位書" : labels.positioning}
        </button>

        {/* 視覺規範/行銷規範按鈕 */}
        <Button
          onClick={onViewReport}
          disabled={!isReportEnabled}
          variant="outline"
          className="flex-1 font-semibold"
          style={{ borderRadius: "50px", borderColor: "#EA580C", color: "#EA580C" }}
        >
          {labels.report}
        </Button>

        {/* AI 顧問按鈕 */}
        <Button
          onClick={onAIAdvisor}
          variant="outline"
          className="flex-1 font-semibold"
          style={{ borderRadius: "50px", borderColor: "#EA580C", color: "#EA580C" }}
        >
          <Sparkles className="h-4 w-4 mr-1" />
          {labels.advisor}
        </Button>
      </div>
    </Card>
  );
}
